function demoData() {
  const G = { "MJ-A": ["名間松柏嶺A區", 23.8341, 120.6452], "PL-B": ["埔里B區", 23.9645, 120.9693],
              "LG-C": ["鹿谷凍頂C區", 23.7449, 120.7871], "YC-D": ["魚池D區", 23.8963, 120.9347], "RA-E": ["仁愛E區", 24.0245, 121.1360] };
  const ds = ["anthracnose", "mite_damage", "white_rot"], sv = ["low", "medium", "high"], recs = [];
  const now = Date.now(), route = [["去程","R",1],["去程","R",2],["去程","R",3],["回程","L",3],["回程","L",2],["回程","L",1]];
  Object.entries(G).forEach(([id, g], gi) => {
    for (let d = 13; d >= 0; d--) route.forEach(([trip, side, st], i) => {
      const sick = Math.random() < 0.12 + gi * 0.05 + (13 - d) * 0.01;
      const s = sick ? sv[Math.floor(Math.random() * 3)] : "none";
      recs.push({ time: now - d * 864e5 - gi * 36e5 + i * 8000, garden_id: id, garden: g[0], lat: g[1], lng: g[2], trip, side, station: st,
        disease: sick ? ds[Math.floor(Math.random() * 3)] : null, confidence: sick ? +(0.6 + Math.random() * 0.35).toFixed(2) : 0,
        severity: s, sprayed: sick });
    });
  });
  const gs = {}; Object.entries(G).forEach(([id, g]) => gs[id] = { name: g[0], lat: g[1], lng: g[2] });
  return { det: recs, gar: gs };
}
async function load() {
  try {
    if (DEMO) { const d = demoData(); ai = d.det.map(r => ({ ...r, src: "ai" })); car = []; gardens = d.gar; }
    else {
      const rq = encodeURIComponent('"$key"');
      const [det, gar, runs] = await Promise.all([
        fetch(`${DB}/detections.json`).then(r => r.json()), fetch(`${DB}/gardens.json`).then(r => r.json()),
        fetch(`${DB}/mbot2/${DEVICE}/runs.json?orderBy=${rq}&limitToLast=100`).then(r => r.json()).catch(() => null)]);
      if (det && det.error) throw new Error(det.error);
      ai = Object.values(det || {}).sort((a, b) => a.time - b.time).map(r => ({ ...r, src: "ai" }));
      car = carRows(runs); gardens = gar || {};
      if (!gardens[CAR_GARDEN.id]) gardens[CAR_GARDEN.id] = { name: CAR_GARDEN.name, lat: CAR_GARDEN.lat, lng: CAR_GARDEN.lng };
    }
    $("status").textContent = (DEMO ? "示範資料｜" : "") + "最後更新 " + new Date().toLocaleTimeString("zh-TW");
    fillGardens(); render();
  } catch (e) { $("status").textContent = "讀取失敗：請確認資料庫網址與規則（" + e.message + "）"; }
}
// 車體回報：每一站一筆；有噴灑＝未分類病灶（車子不知道病害種類與嚴重程度）
function carRows(runs) {
  const out = [];
  Object.values(runs || {}).forEach(run => {
    const st = run && run.stations;
    if (!st || typeof st !== "object") return;
    Object.values(st).forEach(r => out.push({ time: typeof r.time === "number" ? r.time : (run.start || 0),
      garden_id: CAR_GARDEN.id, garden: CAR_GARDEN.name, lat: CAR_GARDEN.lat, lng: CAR_GARDEN.lng,
      trip: r.trip, side: r.side, station: r.station, disease: r.sprayed ? "unclassified" : null, confidence: 0,
      severity: r.sprayed ? "unknown" : "none", sprayed: !!r.sprayed, src: "car" }));
  });
  return out.sort((a, b) => a.time - b.time);
}
function fillGardens() {
  const all = ai.concat(car), sel = $("gardenSel"), cur = sel.value, ids = [...new Set(all.map(r => r.garden_id))];
  sel.innerHTML = '<option value="">全部茶園</option>' + ids.map(id => `<option value="${id}">${(all.find(r => r.garden_id === id) || {}).garden || id}</option>`).join("");
  sel.value = ids.includes(cur) ? cur : "";
}
function rows() {
  const g = $("gardenSel").value, d = +$("rangeSel").value, t0 = d ? Date.now() - d * 864e5 : 0;
  const f = r => (!g || r.garden_id === g) && r.time >= t0, a = ai.filter(f), c = car.filter(f), m = $("srcSel").value;
  return m === "ai" ? a : m === "car" ? c : m === "all" ? a.concat(c).sort((x, y) => x.time - y.time) : (a.length ? a : c);
}

function project(lat, lng) {   // 經緯度 → 示意地圖座標
  const x = (lng - 120.585) / 0.75 * 480, y = (24.2 - lat) / 0.75 * SH;

function drawMap(sick) {

}

function drawHeat(data) {
  const box = $("heat");
  if (!data.length) { box.innerHTML = '<p class="sub">尚無資料</p>'; return; }
  const last = data[data.length - 1], run = data.filter(r => r.garden_id === last.garden_id && last.time - r.time < 300e3);
  const n = Math.max(3, ...run.map(r => r.station));
  let h = `<div class="hg" style="grid-template-columns:40px repeat(${n},1fr)">`;
  ["L", "R"].forEach(side => { h += `<div class="rl">${side} 側</div>`;
    for (let s = 1; s <= n; s++) { const r = run.filter(x => x.side === side && x.station === s).pop();
      h += `<div class="cell" style="background:${r ? SEVC[r.severity] : "#c9d1c6"}">${side}${s}<br>${r ? (r.disease ? dn(r.disease) : "健康") : "—"}</div>`; } });
  box.innerHTML = h + `</div><p class="sub">${last.garden}｜${new Date(last.time).toLocaleString("zh-TW")}</p>`;
}
function drawTable(data) {
  const head = "<tr><th>時間</th><th>茶園</th><th>趟次</th><th>位置</th><th>病害</th><th>信心</th><th>嚴重度</th><th>噴灑</th><th>來源</th></tr>";
  $("tbl").innerHTML = head + [...data].reverse().slice(0, 100).map(r => `<tr><td>${new Date(r.time).toLocaleString("zh-TW")}</td><td>${r.garden}</td><td>${r.trip}</td>
    <td>${r.side}${r.station}</td><td>${r.disease ? dn(r.disease) : "健康"}</td><td>${r.disease ? r.confidence : "—"}</td>
    <td><span class="tag" style="background:${SEVC[r.severity] || "#999"}">${SEVT[r.severity] || r.severity}</span></td><td>${r.sprayed ? "是" : "否"}</td><td>${r.src === "car" ? "車體" : "AI"}</td></tr>`).join("");
}

const COLS = ["time", "garden_id", "garden", "lat", "lng", "trip", "side", "station", "disease", "confidence", "severity", "sprayed", "src"];
const cellv = (r, c) => c === "time" ? new Date(r.time).toISOString() : String(r[c] ?? "");
function save(blob, name) { const a = document.createElement("a"); a.href = URL.createObjectURL(blob); a.download = name; a.click(); }
$("csvBtn").onclick = () => {
  const lines = [COLS.join(",")].concat(rows().map(r => COLS.map(c => '"' + cellv(r, c).replace(/"/g, '""') + '"').join(",")));
  save(new Blob(["\uFEFF" + lines.join("\n")], { type: "text/csv" }), "病蟲害巡檢紀錄.csv");
};
$("xlsBtn").onclick = () => {
  const esc = s => s.replace(/&/g, "&amp;").replace(/</g, "&lt;");
  const html = '<html><head><meta charset="utf-8"></head><body><table border="1"><tr>' + COLS.map(c => "<th>" + c + "</th>").join("") + "</tr>" +
    rows().map(r => "<tr>" + COLS.map(c => "<td>" + esc(cellv(r, c)) + "</td>").join("") + "</tr>").join("") + "</table></body></html>";
  save(new Blob([html], { type: "application/vnd.ms-excel" }), "病蟲害巡檢紀錄.xls");
};
